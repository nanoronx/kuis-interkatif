      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.frequency.value = freq;
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      gain.gain.exponentialRampToValueAtTime(0.00001, audioCtx.currentTime + duration);
      osc.stop(audioCtx.currentTime + duration);
    }

      const display = document.getElementById('auto-next-display');
      const timerDesc = document.getElementById('timer-desc-info');

      if (sec === 0) {
        display.textContent = 'Manual (Klik Guru)';
        display.className = 'font-mono text-xs font-bold text-amber-300 bg-amber-950/80 border border-amber-700/60 px-3 py-1 rounded-lg';
        timerDesc.textContent = 'Saat waktu habis, kunci jawaban ditampilkan. Guru menekan tombol untuk lanjut.';
      } else {
        display.textContent = `Otomatis (${sec} dtk)`;
        display.className = 'font-mono text-xs font-bold text-indigo-300 bg-indigo-950/80 border border-indigo-700/60 px-3 py-1 rounded-lg';
        timerDesc.textContent = `Saat waktu habis, kunci jawaban ditampilkan lalu otomatis berganti soal setelah ${sec} detik.`;
      }

      // Update styling tombol aktif
      document.querySelectorAll('.btn-autonext').forEach(btn => {
        const match = (sec === 0 && btn.textContent.includes('Manual')) || (sec > 0 && btn.textContent.includes(`${sec} detik`));
        if (match) {
          btn.className = "btn-autonext px-3.5 py-1.5 text-xs rounded-lg bg-indigo-600 text-white font-bold shadow";
        } else {
          btn.className = "btn-autonext px-3.5 py-1.5 text-xs rounded-lg bg-slate-700 hover:bg-slate-600 text-slate-300";
        }
      });

      filterQuestions();
    }

    // Refresh Mata Pelajaran
    function refreshMapelDropdown() {
      const currentJenjang = document.getElementById('select-jenjang').value;
      const mapelSelect = document.getElementById('select-mapel');
      
      const filteredByJenjang = questions.filter(q => q.jenjang === currentJenjang);
      const uniqueMapels = [...new Set(filteredByJenjang.map(q => q.mapel))];

      mapelSelect.innerHTML = '';
      if (uniqueMapels.length === 0) {
        mapelSelect.innerHTML = `<option value="">(Belum ada mapel)</option>`;
      } else {
        uniqueMapels.forEach(m => {
          const opt = document.createElement('option');
          opt.value = m;
          opt.textContent = m;
          mapelSelect.appendChild(opt);
        });
      }
    }

    function filterQuestions() {
      const jenjang = document.getElementById('select-jenjang').value;
      const mapel = document.getElementById('select-mapel').value;
      const tingkat = document.getElementById('select-tingkat').value;

      activeQuestions = questions.filter(q => {
        const matchJenjang = q.jenjang === jenjang;
        const matchMapel = !mapel || q.mapel === mapel;
        const matchTingkat = tingkat === 'Semua' || q.tingkat === tingkat;
        return matchJenjang && matchMapel && matchTingkat;
      });

      const autoTxt = autoNextDelay > 0 ? `jeda otomatis ${autoNextDelay} dtk` : 'pergantian manual';
      document.getElementById('question-count-info').textContent = 
        `${activeQuestions.length} soal siap dimainkan, ${timerDuration} detik per soal (${autoTxt}).`;
    }

    document.getElementById('select-jenjang').addEventListener('change', () => {
      refreshMapelDropdown();
      filterQuestions();
    });

    function downloadCSVTemplate() {
      const csvHeader = "Jenjang,Mapel,Tingkat,Pertanyaan,Opsi_A,Opsi_B,Jawaban\n";
      const sampleRow1 = "SMP,Informatika,Sedang,Benua terluas di dunia adalah...,Benua Asia,Benua Afrika,A\n";
      const sampleRow2 = "SMP,Informatika,Mudah,Otak pemroses komputer adalah...,RAM,CPU,B\n";
      
      const blob = new Blob([csvHeader + sampleRow1 + sampleRow2], { type: 'text/csv;charset=utf-8;' });
      const link = document.createElement("a");
      link.href = URL.createObjectURL(blob);
      link.download = "template_soal_kuis.csv";
      link.click();
    }

    async function processBulkUpload() {
      const fileInput = document.getElementById('csv-file-input');
      const pastedText = document.getElementById('raw-csv-paste').value.trim();
      const mode = document.querySelector('input[name="upload-mode"]:checked').value;

      let rawContent = "";
      if (fileInput.files.length > 0) {
        rawContent = await fileInput.files[0].text();
      } else if (pastedText.length > 0) {
        rawContent = pastedText;
      } else {
        alert("Pilih file CSV atau tempel teks dari Excel terlebih dahulu!");
        return;
      }

      const lines = rawContent.split(/\r?\n/).filter(line => line.trim() !== "");
      const parsedQuestions = [];
      let startIndex = lines[0].toLowerCase().includes('jenjang') ? 1 : 0;

      for (let i = startIndex; i < lines.length; i++) {
        const row = lines[i].split(',').map(s => s.trim());
        if (row.length >= 7) {
          parsedQuestions.push({
            id: Date.now() + Math.random(),
            jenjang: row[0],
            mapel: row[1],
            tingkat: row[2],
            pertanyaan: row[3],
            opsiA: row[4],
            opsiB: row[5],
            jawaban: row[6].toUpperCase() === 'B' ? 'B' : 'A'
          });
        }
      }

      if (parsedQuestions.length === 0) {
        alert("Format CSV tidak sesuai! Pastikan memiliki 7 kolom.");
        return;
      }

      questions = mode === 'replace' ? parsedQuestions : [...questions, ...parsedQuestions];
      localStorage.setItem('quiz_questions', JSON.stringify(questions));
      alert(`Berhasil mengimpor ${parsedQuestions.length} soal!`);
      closeUploadModal();
      refreshMapelDropdown();
      filterQuestions();
    }

    function resetToDefaultQuestions() {
      if (confirm("Kembalikan semua soal ke bank soal bawaan?")) {
        questions = [...defaultQuestions];
        localStorage.setItem('quiz_questions', JSON.stringify(questions));
        refreshMapelDropdown();
        filterQuestions();
      }
    }

      modal.classList.toggle('hidden');
      if (!modal.classList.contains('hidden')) {
        renderEditorList();
      }

    function renderEditorList() {
      const container = document.getElementById('editor-question-list');
      const stat = document.getElementById('total-questions-stat');
      container.innerHTML = '';
      stat.textContent = `Total: ${questions.length} Soal Terdaftar`;

      questions.forEach((q, index) => {
        const item = document.createElement('div');
        item.className = 'p-3 bg-slate-800 rounded-xl flex items-center justify-between text-xs border border-slate-700/80';
        item.innerHTML = `
          <div class="pr-3">
            <div class="flex gap-2 items-center mb-1">
              <span class="px-2 py-0.5 rounded bg-indigo-900/60 text-indigo-300 font-bold">${q.jenjang}</span>
              <span class="px-2 py-0.5 rounded bg-slate-700 text-slate-300">${q.mapel}</span>
              <span class="px-2 py-0.5 rounded bg-slate-700 text-slate-400">${q.tingkat}</span>

          </button>
        `;
        container.appendChild(item);
      });
    }

    function deleteQuestion(index) {
      if (confirm("Hapus soal ini?")) {
        questions.splice(index, 1);
        localStorage.setItem('quiz_questions', JSON.stringify(questions));
        renderEditorList();
        refreshMapelDropdown();
        filterQuestions();
      }
    }

    // ================= LOGIKA GAMEPLAY & AUTO NEXT =================
    function startGame() {
      if (activeQuestions.length === 0) {
        alert("Tidak ada soal yang tersedia untuk pilihan filter ini.");
        return;
      }
      currentQuestionIndex = 0;
      document.getElementById('screen-settings').classList.add('hidden');
      document.getElementById('screen-game').classList.remove('hidden');
      loadQuestion();
    }

    function loadQuestion() {
      clearInterval(timerInterval);
      clearInterval(autoNextInterval);

      const q = activeQuestions[currentQuestionIndex];

      // Reset Gaya Zona
      const zoneA = document.getElementById('zone-a');
      const zoneB = document.getElementById('zone-b');
      zoneA.className = "flex-1 border-r-2 border-white/30 flex flex-col justify-between p-8 transition-colors duration-300 bg-blue-600/20";
      zoneB.className = "flex-1 border-l-2 border-white/30 flex flex-col justify-between items-end p-8 transition-colors duration-300 bg-rose-600/20";

      // Reset Tampilan Banner Kunci & Tombol Lanjut
      document.getElementById('reveal-banner').classList.add('hidden');
      document.getElementById('auto-countdown-banner').classList.add('hidden');
      document.getElementById('btn-next-question').classList.add('hidden');
      document.getElementById('timer-box').classList.remove('hidden');

      // Tampilkan Soal & Metadata
      document.getElementById('game-meta-info').textContent = `${q.jenjang} • ${q.mapel.toUpperCase()} (${q.tingkat.toUpperCase()})`;
      document.getElementById('game-progress-info').textContent = `Soal ${currentQuestionIndex + 1} / ${activeQuestions.length}`;
      document.getElementById('game-question-text').textContent = q.pertanyaan;
      document.getElementById('label-optA').textContent = `A. ${q.opsiA}`;
      document.getElementById('label-optB').textContent = `B. ${q.opsiB}`;

      // Hitung Mundur Menjawab Soal
      currentCountdown = timerDuration;
      document.getElementById('game-timer-num').textContent = currentCountdown;

      timerInterval = setInterval(() => {
        currentCountdown--;
        document.getElementById('game-timer-num').textContent = currentCountdown;
        if (currentCountdown <= 3 && currentCountdown > 0) {
          playBeep(880, 0.08);
        }
        if (currentCountdown <= 0) {
          clearInterval(timerInterval);
          revealAnswer();
        }
      }, 1000);
    }

    function revealAnswer() {
      const q = activeQuestions[currentQuestionIndex];
      const zoneA = document.getElementById('zone-a');
      const zoneB = document.getElementById('zone-b');
      const banner = document.getElementById('reveal-banner');
      const keySpan = document.getElementById('reveal-key');

      keySpan.textContent = q.jawaban;
      banner.classList.remove('hidden');
      playSuccessSound();

      // Highlight Sisi Benar (Hijau) & Sisi Salah (Merah Redup)
      if (q.jawaban === 'A') {
        zoneA.className = "flex-1 border-r-4 border-emerald-400 flex flex-col justify-between p-8 transition-all duration-300 bg-emerald-600/60 shadow-inner";
        zoneB.className = "flex-1 border-l-4 border-rose-600 flex flex-col justify-between items-end p-8 transition-all duration-300 bg-rose-900/60 opacity-60";
      } else {
        zoneB.className = "flex-1 border-l-4 border-emerald-400 flex flex-col justify-between items-end p-8 transition-all duration-300 bg-emerald-600/60 shadow-inner";
        zoneA.className = "flex-1 border-r-4 border-rose-600 flex flex-col justify-between p-8 transition-all duration-300 bg-rose-900/60 opacity-60";
      }

      // [LOGIKA GANTI SOAL OTOMATIS]
      if (autoNextDelay > 0) {
        // Mode Otomatis Aktif
        const countdownBanner = document.getElementById('auto-countdown-banner');
        const secondsLeftSpan = document.getElementById('auto-seconds-left');
        countdownBanner.classList.remove('hidden');
        
        autoCountdownLeft = autoNextDelay;
        secondsLeftSpan.textContent = autoCountdownLeft;

        autoNextInterval = setInterval(() => {
          autoCountdownLeft--;
          secondsLeftSpan.textContent = autoCountdownLeft;
          if (autoCountdownLeft <= 0) {
            clearInterval(autoNextInterval);
            advanceToNext();
          }
        }, 1000);
      } else {
        // Mode Manual: Guru Mengklik Tombol
        document.getElementById('btn-next-question').classList.remove('hidden');
      }
    }

    function manualNextQuestion() {
      clearInterval(autoNextInterval);
      advanceToNext();
    }

    function advanceToNext() {
      currentQuestionIndex++;
      if (currentQuestionIndex < activeQuestions.length) {
        loadQuestion();
      } else {
        alert("🎉 Kuis telah selesai!");
        stopGame();
      }
    }

    function stopGame() {
      clearInterval(timerInterval);
      clearInterval(autoNextInterval);
      document.getElementById('screen-game').classList.add('hidden');
      document.getElementById('screen-settings').classList.remove('hidden');
    }
